// Apply for a trekking permit
const response = await fetch('http://localhost:3000/api/v1/tourism/permits/apply', {
  method: 'POST',
  headers: { 'Content-Type': 'application/json' },
  body: JSON.stringify({
    trek_id: 'EBC',
    passport_number: 'P123456',
    nationality: 'Foreign'
  })
});

const { data } = await response.json();
console.log(data.permit_id); // PRMT-9FFE4A8F
import requests

# Apply for a trekking permit
res = requests.post(
    'http://localhost:3000/api/v1/tourism/permits/apply',
    json={
        'trek_id': 'EBC',
        'passport_number': 'P123456',
        'nationality': 'Foreign'
    }
)
data = res.json()['data']
print(data['permit_id'])  # PRMT-9FFE4A8F

Why Nepal API

Built for real-world use

⚡

JWT Authentication

Admin-protected write endpoints secured with JSON Web Tokens. Public read endpoints need no key.

🗄️

MongoDB Atlas Backed

Permits, tickets, jobs, payments, and subscriptions persist in MongoDB Atlas with file fallback for local dev.

🛡️

Validated & Rate Limited

Every endpoint validated with express-validator. 100 req/15 min rate limiting. Helmet security headers.

📦

Docker Ready

Comes with a production Dockerfile. Drop in your env file and ship.

🔄

Live API Fallbacks

External integrations (NRB forex, OpenWeatherMap) gracefully fall back to realistic cached data if unavailable.

📋

Structured Logging

Winston JSON logs + Morgan HTTP access logs. Production-mode uses combined Apache format.
